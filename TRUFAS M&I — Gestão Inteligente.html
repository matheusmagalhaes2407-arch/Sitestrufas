Configuração do Banco de Dados Supabase

O sistema opera por padrão em Modo Local / Demo armazenando no navegador. Para ativar a sincronização em tempo real entre parceiros em dispositivos diferentes, insira as credenciais do seu projeto Supabase abaixo.

Script SQL de Criação das Tabelas (Supabase):

-- SCRIPT DE CRIAÇÃO TRUFAS M&I
CREATE TABLE IF NOT EXISTS produtos (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nome TEXT NOT NULL,
  descricao TEXT,
  preco_venda NUMERIC NOT NULL,
  custo_producao NUMERIC NOT NULL,
  estoque_atual INT DEFAULT 0,
  estoque_minimo INT DEFAULT 5,
  ativo BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE IF NOT EXISTS vendas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  data_venda TIMESTAMPTZ DEFAULT now(),
  parceiro TEXT NOT NULL,
  cliente TEXT,
  forma_pagamento TEXT NOT NULL,
  total NUMERIC NOT NULL,
  observacao TEXT
);

CREATE TABLE IF NOT EXISTS despesas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  data_despesa TIMESTAMPTZ DEFAULT now(),
  descricao TEXT NOT NULL,
  categoria TEXT NOT NULL,
  valor NUMERIC NOT NULL,
  carteira TEXT NOT NULL,
  parceiro TEXT NOT NULL
);
